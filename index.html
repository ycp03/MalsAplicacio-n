    function assignPlayerToSlot(slotId, playerId) {

      const player = getPlayerById(playerId);

      if (!player || !player.available) return;


      const previousSlotOfPlayer = getAssignedSlotForPlayer(playerId);

      const displacedPlayerId = state.lineup[slotId];


      // Si el jugador ya estaba en otro puesto del campo, intercambiamos ambos puestos

      if (previousSlotOfPlayer && previousSlotOfPlayer !== slotId) {

        state.lineup[previousSlotOfPlayer] = displacedPlayerId || null;

      }


      state.lineup[slotId] = playerId;

      saveState();

      renderAll();

      closeSlotModal();

    }

    function quickAssignFromBench(playerId) {

      const player = getPlayerById(playerId);

      if (!player || !player.available) return;

      if (getAssignedSlotForPlayer(playerId)) return;


      // Buscar primero un hueco libre compatible con su posición

      let targetSlot = PITCH_SLOTS.find(s => !state.lineup[s.id] && s.compatible.includes(player.position));

      // Si no hay compatible libre, buscar cualquier hueco libre (evitando poner a un jugador de campo en POR salvo que no quede otra)

      if (!targetSlot) {

        targetSlot = PITCH_SLOTS.find(s => !state.lineup[s.id] && s.id !== 'POR') || PITCH_SLOTS.find(s => !state.lineup[s.id]);

      }


      if (targetSlot) {

        assignPlayerToSlot(targetSlot.id, playerId);

        showToast(`${player.name} entra de titular en ${targetSlot.label}.`, 'emerald');

      } else {

        showToast('El 7 inicial ya está completo. Toca una posición en el campo para sustituir.', 'amber');

      }

    }


    function autoFillLineup() {

      const available = state.players.filter(p => p.available);

      if (available.length === 0) {

        showToast('No hay jugadores convocados disponibles.', 'amber');

        return;

      }


      const usedIds = getStarterIdsSet();

      let filled = 0;


      // 1ª pasada: rellenar puestos vacíos con jugadores ideales para cada demarcación

      for (const slot of PITCH_SLOTS) {

        if (state.lineup[slot.id]) continue;

        const candidate = available.find(p => !usedIds.has(p.id) && slot.compatible.includes(p.position));

        if (candidate) {

          state.lineup[slot.id] = candidate.id;

          usedIds.add(candidate.id);

          filled++;

        }

      }


      // 2ª pasada: si aún quedan puestos vacíos y hay convocados libres, completarlos

      for (const slot of PITCH_SLOTS) {

        if (state.lineup[slot.id]) continue;

        const candidate = available.find(p => !usedIds.has(p.id));

        if (candidate) {

          state.lineup[slot.id] = candidate.id;

          usedIds.add(candidate.id);

          filled++;

        }

      }


      saveState();

      renderAll();

      if (filled > 0) {

        showToast(`7 inicial completado (+${filled} ${filled === 1 ? 'jugador' : 'jugadores'}).`, 'emerald');

      } else {

        showToast('El 7 inicial ya tiene todos los puestos posibles cubiertos.', 'amber');

      }

    }

    function applyWhatsAppList() {

      const rawText = document.getElementById('whatsapp-textarea').value;

      if (!rawText.trim()) {

        showToast('Pega primero el texto con los jugadores asistentes.', 'amber');

        return;

      }


      const replaceRest = document.getElementById('whatsapp-replace-check').checked;

      const lines = rawText.split(/\r?\n/).map(l => normalizeText(l)).filter(Boolean);

      const fullNormalized = normalizeText(rawText);


      // Extraer números explícitos precedidos de # o entre paréntesis para evitar confundir "1. Nombre" de una lista numerada

      const explicitDorsals = new Set();

      const dorsalMatches = rawText.matchAll(/(?:#|\()\s*(\d{1,2})\s*\)?/g);

      for (const m of dorsalMatches) {

        explicitDorsals.add(m[1]);

      }


      const matchedIds = new Set();


      state.players.forEach(player => {

        const normName = normalizeText(player.name);

        const nameTokens = normName.split(/\s+/);

        const firstName = nameTokens[0];


        // 1. Coincidencia por nombre completo

        if (fullNormalized.includes(normName)) {

          matchedIds.add(player.id);

          return;

        }


        // 2. Coincidencia por dorsal explícito (#75, (75))

        if (player.dorsal !== '-' && explicitDorsals.has(player.dorsal)) {

          matchedIds.add(player.id);

          return;

        }


        // 3. Coincidencia inteligente por línea (ej. "Alvaro Gorjon" vs "Alvaro", "Luis", "Jorge", "Hugo", "Adrian")

        for (const line of lines) {

          // Limpiar numeración inicial tipo "1.", "2)", "- "

          const cleanLine = line.replace(/^\d+[\.\)\-\s]+/, '').trim();

          if (!cleanLine) continue;


          if (firstName === 'alvaro') {

            if (normName.includes('gorjon') && cleanLine.includes('gorjon')) {

              matchedIds.add(player.id);

              break;

            }

            if (!normName.includes('gorjon') && /\balvaro\b/.test(cleanLine) && !cleanLine.includes('gorjon')) {

              matchedIds.add(player.id);

              break;

            }

          } else if (normName.includes('otaduy')) {

            if (cleanLine.includes(firstName)) {

              matchedIds.add(player.id);

              break;

            }

          } else if (normName.includes('gorjon')) {

            if (cleanLine.includes(firstName) || (cleanLine.includes('gorjon') && !cleanLine.includes('alvaro'))) {

              matchedIds.add(player.id);

              break;

            }

          } else {

            const firstRegex = new RegExp(`\\b${firstName}\\b`);

            if (firstRegex.test(cleanLine)) {

              matchedIds.add(player.id);

              break;

            }

          }

        }

      });


      if (matchedIds.size === 0) {

        showToast('No se reconoció ningún jugador de la plantilla en el texto.', 'amber');

        return;

      }


      state.players.forEach(p => {

        if (matchedIds.has(p.id)) {

          p.available = true;

        } else if (replaceRest) {

          p.available = false;

        }

      });


      saveState();

      renderAll();

      closeWhatsAppModal();

      showToast(`Convocatoria actualizada: ${matchedIds.size} jugadores detectados.`, 'emerald');

    }
